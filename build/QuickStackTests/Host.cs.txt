public class ItemData
{
    public enum ItemType { None, Material, OneHandedWeapon, Bow, Shield, Helmet, Chest, Ammo, Customization,
        Legs, Hands, Trophy, TwoHandedWeapon, Torch, Misc, Shoulder, Utility, Tool, Attach_Atgeir, Fish,
        TwoHandedWeaponLeft, AmmoNonEquipable, Consumable, Trinket }
    public class Shared { public string m_name = "base"; public int m_maxStackSize = 1; public ItemType m_itemType = ItemType.Material; }
    public Shared m_shared = new();
    public int m_stack = 1, m_quality = 1, m_worldLevel;
    public bool m_cheated, m_equipped, Favorite, Special, MetadataAllowed = true;
    public Vector2i m_gridPos = new(0, 1);
    public ItemData Clone() => (ItemData)MemberwiseClone();
}
public class Player
{
    public bool Destroyed;
    public bool IsItemEquiped(ItemData item) => item.m_equipped;
    // Model Unity's destroyed-object equality separately from CLR null.
    public static bool operator ==(Player? a, Player? b)
    {
        bool aNull = ReferenceEquals(a, null) || a.Destroyed;
        bool bNull = ReferenceEquals(b, null) || b.Destroyed;
        return aNull || bNull ? aNull && bNull : ReferenceEquals(a, b);
    }
    public static bool operator !=(Player? a, Player? b) => !(a == b);
    public override bool Equals(object? other) => ReferenceEquals(this, other);
    public override int GetHashCode() => base.GetHashCode();
}
public class Inventory
{
    public List<ItemData> m_inventory = new();
    public int Width = 3, Height = 2, Changes, MoveCalls;
    public bool Veto;
    public Action? AfterMove;
    public int GetWidth() => Width;
    public int GetHeight() => Height;
    public List<ItemData> GetAllItems() => m_inventory;
    public ItemData? GetItemAt(int x, int y) => m_inventory.FirstOrDefault(i => i.m_gridPos == new Vector2i(x, y));
    public void Changed() => Changes++;
    // Fake native positional move: inventory ownership and source quantities are
    // mutated together. This does not simulate Harmony or native Unity callbacks.
    public bool MoveItemToThis(Inventory from, ItemData source, int requested, int x, int y)
    {
        MoveCalls++;
        if (Veto || !from.m_inventory.Contains(source)) return false;
        var target = GetItemAt(x, y);
        int moved = Math.Min(source.m_stack, Math.Min(requested, target == null
            ? source.m_shared.m_maxStackSize : target.m_shared.m_maxStackSize - target.m_stack));
        if (moved <= 0) return false;
        if (target == null) { target = source.Clone(); target.m_stack = 0; target.m_gridPos = new(x, y); m_inventory.Add(target); }
        target.m_stack += moved;
        source.m_stack -= moved;
        if (source.m_stack == 0) from.m_inventory.Remove(source);
        AfterMove?.Invoke();
        return true;
    }
}

public class Boundaries
{
    protected static IEnumerable<string?> GetRestockTargetLookupTokens(ItemData item) => new[] { item.m_shared.m_name };
    protected static bool IsRegularActionItem(Inventory inventory, ItemData item) => !item.Special && item.m_gridPos.y > 0;
    protected static bool IsRegularActionItem(Player player, Inventory inventory, ItemData item) => IsRegularActionItem(inventory, item);
    protected static bool IsFavoriteProtected(Player player, Inventory inventory, ItemData item) => item.Favorite;
    protected static bool CanUseContainerActionStacking(ItemData item) => item.MetadataAllowed;
    protected static bool IsTrustedCustomDataStackingItem(ItemData item) => false;
    protected static bool IsEpicLootStackingItem(ItemData item) => false;
    protected static bool CanMergeEpicLootStacks(ItemData target, ItemData source) => false;
    protected static bool CanShareInventoryStack(ItemData target, ItemData source) =>
        target.m_shared.m_name == source.m_shared.m_name && target.m_quality == source.m_quality &&
        target.m_worldLevel == source.m_worldLevel && target.m_cheated == source.m_cheated;
    protected static void NotifyInventoryChanged(Inventory inventory) => inventory.Changed();
    protected static int CompareGridOrder(Vector2i a, Vector2i b) => a.y != b.y ? a.y.CompareTo(b.y) : a.x.CompareTo(b.x);
}

public static class Checks
{
    private static int Count;
    private static void Check(bool condition, string message) { Count++; if (!condition) throw new Exception(message); }
    private static ItemData Item(string name = "base", int max = 1, int count = 1, int x = 0, int y = 1) => new()
    { m_shared = new() { m_name = name, m_maxStackSize = max }, m_stack = count, m_gridPos = new(x, y) };
    public static void Run()
    {
        foreach (var (name, eligible, transfer) in new (string, Func<ItemData, bool>, Func<Inventory, Inventory, int>)[]
            { ("Slots", Slots.Eligible, Slots.Transfer), ("Actions", Actions.Eligible, Actions.Transfer) })
        {
            Check(eligible(Item()), name + ": single material must be a candidate");
            Check(!eligible(null!), name + ": null item");
            Check(!eligible(new() { m_shared = null! }), name + ": null shared data");
            foreach (ItemData.ItemType type in Enum.GetValues<ItemData.ItemType>())
            {
                var item = Item(); item.m_shared.m_itemType = type;
                Check(eligible(item) == (type == ItemData.ItemType.Material), name + ": single item type " + type);
                item.m_shared.m_maxStackSize = 20;
                Check(eligible(item), name + ": preserve stackable item type " + type);
            }
            foreach (Action<ItemData> protect in new Action<ItemData>[] {
                i => i.Favorite = true, i => i.Special = true, i => i.m_gridPos = new(0, 0),
                i => i.MetadataAllowed = false, i => i.m_equipped = true })
            { var item = Item(); protect(item); Check(!eligible(item), name + ": source protection"); }
            Check(!eligible(Item(max: 0)), name + ": invalid max size remains excluded");
            var source = Item(); var seed = Item(x: 0, y: 0);
            var from = new Inventory(); from.m_inventory.Add(source);
            var to = new Inventory(); to.m_inventory.Add(seed);
            Check(transfer(from, to) == 1, name + ": same material moves to a free cell");
            Check(from.m_inventory.Count == 0 && seed.m_stack == 1 && to.m_inventory.Count == 2 &&
                to.GetItemAt(1, 0)?.m_stack == 1 && !ReferenceEquals(to.GetItemAt(1, 0), source), name + ": source removed once, seed intact, new stack top first");
            Check(transfer(from, to) == 0, name + ": repeat cannot duplicate moved item");
            foreach (string blocked in new[] { "empty", "different", "full", "veto" })
            {
                source = Item(); from = new(); from.m_inventory.Add(source);
                to = new() { Width = blocked == "full" ? 1 : 3, Height = 1, Veto = blocked == "veto" };
                if (blocked != "empty") to.m_inventory.Add(Item(blocked == "different" ? "other" : "base", y: 0));
                Check(transfer(from, to) == 0 && from.m_inventory.Contains(source) && source.m_stack == 1, name + ": no loss when " + blocked);
            }
            from = new(); from.m_inventory.Add(Item()); from.m_inventory.Add(Item(x: 1));
            to = new() { Width = 2, Height = 1 }; to.m_inventory.Add(Item(y: 0));
            Check(transfer(from, to) == 1 && from.m_inventory.Count == 1 && to.m_inventory.Sum(i => i.m_stack) == 2, name + ": only available cell used");
            from = new(); from.m_inventory.Add(Item("wood", 20, 20));
            seed = Item("wood", 20, 20, y: 0); to = new(); to.m_inventory.Add(seed);
            Check(transfer(from, to) == 1 && seed.m_stack == 20 && to.GetItemAt(1, 0)?.m_stack == 20, name + ": full ordinary stack spills into empty cell");
            from = new(); from.m_inventory.Add(Item("wood", 20, 15));
            seed = Item("wood", 20, 10, y: 0); to = new(); to.m_inventory.Add(seed);
            Check(transfer(from, to) == 1 && seed.m_stack == 20 && to.GetItemAt(1, 0)?.m_stack == 5, name + ": ordinary partial stack filled before free cell");
        }
        foreach (string condition in new[] { "allowed", "favorite", "equipped", "special", "metadata", "different", "empty", "full" })
        {
            var source = Item(); source.Favorite = condition == "favorite"; source.m_equipped = condition == "equipped";
            source.Special = condition == "special"; source.MetadataAllowed = condition != "metadata";
            var from = new Inventory(); from.m_inventory.Add(source);
            var to = new Inventory { Width = condition == "full" ? 1 : 3, Height = 1 };
            if (condition != "empty") to.m_inventory.Add(Item(condition == "different" ? "other" : "base", y: 0));
            Check(Actions.Muc(from, to) == (condition == "allowed"), "MUC candidate dispatch: " + condition);
            if (condition == "allowed") Check(Actions.Sent is { } request && request.Target == null &&
                ReferenceEquals(request.Source, source) && request.Cell == new Vector2i(1, 0) && request.Amount == 1 &&
                from.m_inventory.Contains(source), "MUC requests one new cell through adapter without locally removing source");
        }
        CheckSurplus();
        Console.WriteLine($"{Count} quick-stack checks passed for both mods; native transfer and MUC request boundaries are doubles, no game execution.");
    }

    private static void Rules(string text) => InventoryDefinitions.RestockTargetStackLimits = RestockTargetLimitCore.Parse(text);
    private static (Inventory From, Inventory To, ItemData Source) Surplus(int count = 150, int seed = 180)
    {
        var source = Item("Stone", 200, count); source.Favorite = true;
        var from = new Inventory(); from.m_inventory.Add(source);
        var to = new Inventory(); to.m_inventory.Add(Item("Stone", 200, seed, y: 0));
        return (from, to, source);
    }
    private static void CheckSurplus()
    {
        foreach (var (name, transfer, stale) in new (string, Func<Inventory, Inventory, int>, Func<Inventory, Inventory, int>)[]
            { ("Slots", Slots.Transfer, Slots.StaleTransfer), ("Actions", Actions.Transfer, Actions.StaleTransfer) })
        {
            foreach (string mode in new[] { "Existing", "IncludeEmpty" })
            {
                Rules("Stone: 100 | " + mode);
                var (from, to, source) = Surplus();
                Check(transfer(from, to) == 1 && source.m_stack == 100 && to.m_inventory.Sum(i => i.m_stack) == 230,
                    name + ": per-stack surplus fills partial stack then empty cell: " + mode);
                Check(to.GetItemAt(0, 0)!.m_stack == 200 && to.GetItemAt(1, 0)!.m_stack == 30, name + ": partial/empty budgets");
                var nextChest = Surplus().To;
                Check(transfer(from, nextChest) == 0 && source.m_stack == 100, name + ": subsequent chest preserves target");
            }
            foreach (string rule in new[] { "", "Stone: 100 | Off", "Stone: invalid | Existing", "Stone: 0 | Existing" })
            {
                Rules(rule); var (from, to, source) = Surplus();
                Check(transfer(from, to) == 0 && stale(from, to) == 0 && source.m_stack == 150 && to.MoveCalls == 0,
                    name + ": absent/Off/invalid target protects favorites even with stale candidates: " + rule);
                source.Favorite = false;
                Check(transfer(from, to) == 1 && !from.m_inventory.Contains(source), name + ": nonfavorites unchanged: " + rule);
            }
            Rules("Stone: 100 | Existing");
            foreach (int count in new[] { 80, 100 })
            {
                var (from, to, source) = Surplus(count);
                Check(stale(from, to) == 0 && source.m_stack == count && to.MoveCalls == 0, name + ": no surplus below/at target");
            }
            var pair = Surplus(); var other = Item("Stone", 200, 80, x: 1); other.Favorite = true; pair.From.m_inventory.Add(other);
            Check(transfer(pair.From, pair.To) == 1 && pair.Source.m_stack == 100 && other.m_stack == 80, name + ": each favorite protected independently, not inventory total");
            foreach (Action<ItemData> protect in new Action<ItemData>[] { i => i.m_equipped = true, i => i.Special = true,
                i => i.m_gridPos = new(0, 0), i => i.MetadataAllowed = false })
            {
                pair = Surplus(); protect(pair.Source);
                Check(stale(pair.From, pair.To) == 0 && pair.Source.m_stack == 150, name + ": enabled target cannot bypass source guard");
            }
            pair = Surplus(); pair.To.Width = 1; pair.To.Height = 1;
            Check(transfer(pair.From, pair.To) == 1 && pair.Source.m_stack == 130, name + ": partial first chest");
            Check(transfer(pair.From, Surplus(seed: 200).To) == 1 && pair.Source.m_stack == 100, name + ": next chest receives remaining surplus only");
            foreach (string changed in new[] { "Stone: 140 | Existing", "Stone: 100 | Off" })
            {
                Rules("Stone: 100 | Existing"); pair = Surplus();
                pair.To.AfterMove = () => Rules(changed);
                Check(transfer(pair.From, pair.To) == 1 && pair.Source.m_stack == 130 && pair.To.MoveCalls == 1,
                    name + ": rule change after first move stops subsequent writes: " + changed);
            }
            Rules("Stone: 100 | Existing"); pair = Surplus(seed: 200); pair.To.Veto = true;
            Check(transfer(pair.From, pair.To) == 0 && pair.Source.m_stack == 150, name + ": native rejection preserves source");
            pair = Surplus(seed: 200); pair.Source.m_shared.m_maxStackSize = 80;
            Check(transfer(pair.From, pair.To) == 1 && pair.Source.m_stack == 80, name + ": effective target clamps to current max stack");
        }
        Rules("Stone: 100 | Existing");
        foreach (var transfer in new Func<Player, Inventory, Inventory, int>[] { Slots.WithPlayer, Actions.WithPlayer })
        {
            var pair = Surplus(); var player = new Player();
            pair.To.AfterMove = () => player.Destroyed = true;
            Check(transfer(player, pair.From, pair.To) == 1 && pair.Source.m_stack == 130 && pair.To.MoveCalls == 1,
                "destroyed player must not turn a restricted quick-stack move into an unrestricted move");
        }
        var muc = Surplus();
        Check(Actions.Muc(muc.From, muc.To) && Actions.Sent!.Value.Amount == 20 && muc.Source.m_stack == 150,
            "MUC: partial request capped without local mutation");
        // Simulate the adapter's accepted response before requesting the next move.
        var sent = Actions.Sent!.Value;
        muc.To.MoveItemToThis(muc.From, muc.Source, sent.Amount, sent.Cell.x, sent.Cell.y);
        Check(Actions.Muc(muc.From, muc.To) && Actions.Sent!.Value.Amount == 30 && Actions.Sent.Value.Target == null,
            "MUC: next empty cell request only gets remaining surplus");
        Rules("Stone: 100 | Off");
        Check(!Actions.Muc(muc.From, muc.To) && muc.Source.m_stack == 130, "MUC: Off prevents next dispatch");
        Rules("Stone: 100 | IncludeEmpty");
        Check(Actions.Muc(muc.From, muc.To) && Actions.Sent!.Value.Amount == 30, "MUC: IncludeEmpty shares surplus policy");
        sent = Actions.Sent!.Value; muc.To.MoveItemToThis(muc.From, muc.Source, sent.Amount, sent.Cell.x, sent.Cell.y);
        Check(!Actions.Muc(muc.From, muc.To) && muc.Source.m_stack == 100, "MUC: response reaching target stops dispatch");
        Rules("");
    }
}
