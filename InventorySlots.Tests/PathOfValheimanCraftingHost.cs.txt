public sealed partial class InventorySlotsPlugin {
/* PRODUCTION */
private static int _uiLocalizationVersion;
private static int warnings;
private static bool ownsUi;
private static bool vneiOpen;
private static bool IsCraftingPrimaryTabSelected(InventoryGui gui) => gui.PrimarySelected;
private static bool IsVneiCraftingTabActive(InventoryGui gui) => vneiOpen;
private static bool ShouldShowCraftingPanelRedesign(InventoryGui? gui) => ownsUi;
private static void WarnPathOfValheimanCrafting(Exception error) => warnings++;
private static void Verify(bool value, string label) { if (!value) throw new Exception(label); }
public static void Check() {
    var gui = InventoryGui.instance = new InventoryGui();
    var parent = gui.m_tabCraft.transform.parent = new Transform();
    var tab = new Button { interactable = false }; tab.transform.parent = parent;
    Verify(!IsPathOfValheimanSalvageTab(gui), "PoV absent cannot claim the tab");
    bool salvaging = true;
    _povSalvaging = () => salvaging; _povSalvageTab = () => tab;
    Verify(IsPathOfValheimanSalvageTab(gui), "active native PoV selection recognized");
    vneiOpen = true;
    Verify(!IsPathOfValheimanSalvageTab(gui), "VNEI owns its panel despite stale PoV selection");
    vneiOpen = false;
    gui.PrimarySelected = true;
    Verify(!IsPathOfValheimanSalvageTab(gui), "stale PoV state cannot override Craft/Upgrade");
    gui.PrimarySelected = false; tab.gameObject.activeInHierarchy = false;
    Verify(!IsPathOfValheimanSalvageTab(gui), "hidden PoV tab not selected");
    tab.gameObject.activeInHierarchy = true; tab.transform.parent = new Transform();
    Verify(!IsPathOfValheimanSalvageTab(gui), "old GUI tab not adopted");
    tab.transform.parent = parent;
    Verify(!IsPathOfValheimanSalvageTab(new InventoryGui()), "old GUI callback ignored");
    _povSalvaging = () => throw new Exception("PoV callback");
    Verify(!IsPathOfValheimanSalvageTab(gui) && warnings == 1, "bad callback falls back");
    _povSalvaging = () => salvaging;
    ownsUi = false; bool categories = true; SuppressPathOfValheimanCraftCategories(ref categories);
    Verify(categories && SkipPathOfValheimanQueueDisplay(gui), "unowned UI keeps PoV layout");
    ownsUi = true; SuppressPathOfValheimanCraftCategories(ref categories);
    Verify(!categories && !SkipPathOfValheimanQueueDisplay(gui), "Slots layout avoids duplicate queue and categories");

    var enabledRecipe = new Recipe(); var disabledRecipe = new Recipe();
    var chests = new List<Container> { new Container() }; int countCalls = 0;
    Verify(GetPathOfValheimanCraftingStock(Player.m_localPlayer, "Wood", enabledRecipe, 8) == 8, "absent PoV keeps existing count");
    _povChestsFor = recipe => recipe == enabledRecipe;
    _povChests = player => chests;
    _povCountEverywhere = (player, name, found) => {
        Verify(found == chests && name == "Wood", "PoV filtered chest list and shared name passed intact"); countCalls++; return 20;
    };
    Verify(GetPathOfValheimanCraftingStock(Player.m_localPlayer, "Wood", disabledRecipe, 8) == 8 && countCalls == 0,
        "displayed recipe's mastery gates count, independent of selected recipe");
    Verify(GetPathOfValheimanCraftingStock(Player.m_localPlayer, "Wood", enabledRecipe, 8) == 20,
        "total includes inventory once, not 8+20");
    Verify(GetPathOfValheimanCraftingStock(Player.m_localPlayer, "Wood", enabledRecipe, 25) == 25,
        "larger optional provider total preserved");
    enabledRecipe.m_requireOnlyOneIngredient = true;
    Verify(GetPathOfValheimanCraftingStock(Player.m_localPlayer, "Wood", enabledRecipe, 8) == 8 && countCalls == 2,
        "one-ingredient recipes retain PoV's native exclusion");
    enabledRecipe.m_requireOnlyOneIngredient = false;
    _povCountEverywhere = (_, _, _) => throw new Exception("chests");
    Verify(GetPathOfValheimanCraftingStock(Player.m_localPlayer, "Wood", enabledRecipe, 8) == 8, "provider failure keeps existing count");

    var first = new ItemData(); var second = new ItemData(); var wood = new ItemDrop();
    var pair = new InventoryGui.RecipeDataPair { ItemData = first, Recipe = enabledRecipe };
    bool rune = false; int previews = 0;
    _povRunecarver = _ => rune;
    _povText = (_, fallback) => fallback;
    _povSalvageReturns = (item, recipe) => { previews++; return new() { (wood, item == first ? 7 : 11) }; };
    var original = GetPathOfValheimanPreview(pair);
    Verify(original.Returns.Single().Count == 7 && original.Summary.Contains("Wood x7"), "actual item return values displayed");
    Verify(ReferenceEquals(original, GetPathOfValheimanPreview(pair)) && previews == 1, "steady hover reuses preview");
    pair.ItemData = second;
    Verify(GetPathOfValheimanPreview(pair).Returns.Single().Count == 11 && previews == 2, "same recipe retains distinct item identity");
    pair.ItemData = first; pair.Recipe = disabledRecipe;
    GetPathOfValheimanPreview(pair); Verify(previews == 3, "recipe replacement invalidates preview");
    _uiLocalizationVersion++; GetPathOfValheimanPreview(pair); Verify(previews == 4, "language change invalidates preview");
    Time.unscaledTime = 0.3f; GetPathOfValheimanPreview(pair); Verify(previews == 5, "node/return tuning refreshed on bounded timer");
    ObjectDB.instance.Items["PT_AzuriteDust"] = new GameObject { Item = wood };
    _povDisenchantReturns = (ItemData item, out int a, out int d) => { a = 3; d = 0; };
    _povSocketContents = _ => new[] { "gem", "", "gem2" };
    rune = true;
    var disenchant = GetPathOfValheimanPreview(pair);
    Verify(disenchant.Disenchant && disenchant.Returns.Single().Count == 3 && disenchant.Summary.Contains("cannot be recovered (2)"),
        "station switch immediately changes returns and preserves socket-loss warning");
    rune = false;
    for (int i = 0; i < 130; i++) { pair.ItemData = new ItemData(); GetPathOfValheimanPreview(pair); }
    Verify(PoVCraftingPreviews.Count <= 128, "preview cache remains bounded");
    ClearPathOfValheimanCraftingPreviews(); Verify(PoVCraftingPreviews.Count == 0, "list invalidation clears item references");
}
}
