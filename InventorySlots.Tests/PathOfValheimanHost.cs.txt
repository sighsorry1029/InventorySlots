public static void Check() {
    var player = Player.m_localPlayer = new Player();
    var normal = new ItemData { m_gridPos = new(1,1) };
    var equipped = new ItemData { m_gridPos = new(2,1), m_equipped = true };
    var favorite = new ItemData { m_gridPos = new(3,1), Favorite = true };
    var items = player.Inventory.m_inventory;
    items.AddRange(new[] { normal, equipped, favorite, new ItemData {m_gridPos = new(0,0)}, new ItemData {m_gridPos = new(0,4)}, new ItemData {m_gridPos = new(0,9)}, new ItemData {m_gridPos = new(0,10)} });
    var movable = new List<ItemData>(items) { new ItemData {m_gridPos = new(7,1)} };
    FilterPathOfValheimanMovableItems(player, movable);
    Verify(movable.SequenceEqual(new[] { normal }), "movable retains only present, unprotected regular items");
    Verify(!SortPathOfValheimanInventory(player), "PoV direct sort suppressed");
    Verify(!sources.Contains(equipped) && !destinations.Contains(equipped.m_gridPos), "equipped source AND destination preserved");
    Verify(!sources.Contains(favorite) && !destinations.Contains(favorite.m_gridPos), "favorite source AND destination preserved");
    Verify(sources.SequenceEqual(new[] { normal }), "sort excludes tail, hotbar and locked cells");
    InventoryGui.instance = null; sources.Clear();
    Verify(!SortPathOfValheimanInventory(player) && sources.Count == 0, "missing GUI does not invoke unsafe sort");

    var tooltip = UITooltip.m_current = new UITooltip();
    var root = UITooltip.m_tooltip = new GameObject { activeSelf = false }; // Native hover delay.
    var body = new GameObject(); body.transform.Parent = root.transform;
    Verify(!TryYieldToPathOfValheimanTooltip(tooltip), "PoV absent retains Slots tooltip");
    bool wanted = true;
    _pathOfValheimanTooltipWanted = () => wanted;
    _pathOfValheimanTooltipOwners = new(); _pathOfValheimanTooltipOwners.Add(tooltip, normal);
    _pathOfValheimanCurrentTooltip = () => UITooltip.m_current!;
    _pathOfValheimanTooltipRoot = () => UITooltip.m_tooltip!;
    _pathOfValheimanTooltipBody = () => body;
    _pathOfValheimanRefreshTooltip = () => refreshes++;
    Verify(TryYieldToPathOfValheimanTooltip(tooltip) && refreshes == 0, "existing PoV card owns even delayed native root without rebuild");
    OnVanillaTooltipHidden(); // PoV suppressed original Hide while Alt held.
    Verify(TryYieldToPathOfValheimanTooltip(tooltip) && refreshes == 0, "Alt pinned Hide cancellation preserves handlers");
    wanted = false; Verify(!TryYieldToPathOfValheimanTooltip(tooltip), "live Off returns Slots ownership");
    wanted = true; Verify(TryYieldToPathOfValheimanTooltip(tooltip) && refreshes == 1, "live On refreshes once");
    TryYieldToPathOfValheimanTooltip(tooltip); Verify(refreshes == 1, "steady hover does not rebuild");
    tooltip.IsItem = false; Verify(!TryYieldToPathOfValheimanTooltip(tooltip), "noninventory source not claimed");
    tooltip.IsItem = true; body.transform.Parent = new Transform();
    Verify(!TryYieldToPathOfValheimanTooltip(tooltip), "failed/misparented PoV rendering keeps text fallback");
    _pathOfValheimanTooltipWanted = () => throw new Exception("external callback");
    Verify(!TryYieldToPathOfValheimanTooltip(tooltip), "callback error keeps fallback");
    UITooltip.m_current = null; UITooltip.m_tooltip = null; OnVanillaTooltipHidden();
    Verify(_pathOfValheimanTooltipSource == null, "real hide clears references");
    _pathOfValheimanTreeOpen = () => true; Verify(IsPathOfValheimanTreeOpen(), "open tree gates guide");
    _pathOfValheimanTreeOpen = () => throw new Exception("tree getter");
    Verify(!IsPathOfValheimanTreeOpen() && _pathOfValheimanTreeOpen == null, "failed external tree getter disabled once");
}
